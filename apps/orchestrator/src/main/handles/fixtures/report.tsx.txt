function LapTable() {
  return (
    <Table
      title="Lap times"
      columns={[
        { key: 'lap', label: 'Lap' },
        { key: 'time', label: 'Time', align: 'right' },
        { key: 'best', label: 'Best so far' },
        { key: 'ghost', label: 'Against the ghost', align: 'right' }
      ]}
      rows={laps}
    />
  )
}

function Split({ label, value, pct }: { label: string; value: string; pct: number }) {
  return (
    <div style={{ display: 'flex', flexDirection: 'column', gap: 4 }}>
      <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: 12, color: 'var(--ink-2)' }}>
        <span>{label}</span>
        <span style={{ fontVariantNumeric: 'tabular-nums', color: 'var(--ink)' }}>{value}</span>
      </div>
      <div style={{ height: 6, borderRadius: 3, background: 'var(--elevated)', overflow: 'hidden' }}>
        <div style={{ width: `${pct}%`, height: '100%', background: 'var(--accent)' }} />
      </div>
    </div>
  )
}
