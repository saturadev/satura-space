using System.Collections;
using SaturaSpace;
using UnityEngine;

public class JumpPad : MonoBehaviour
{
    public Rigidbody ball;
    public Transform pad;
    public Transform ledge;
    public Renderer goal;
    public TrailRenderer trail;

    static JumpPad _instance;
    static readonly Color Idle = new Color(0.9f, 0.55f, 0.12f);
    static readonly Color Hit = new Color(0.15f, 0.95f, 0.3f);
    static readonly Color Miss = new Color(1f, 0.18f, 0.12f);

    void Awake() { _instance = this; SetGoal(Idle); }

    static Vector3 LaunchVelocity()
    {
        return new Vector3(0f, 8.5f, 5.1f);
    }

    void SetGoal(Color c)
    {
        var m = goal.material;
        m.SetColor("_BaseColor", c * 0.6f);
        m.SetColor("_EmissionColor", c * 1.3f);
    }

    [SaturaHandle("Launches the ball from the jump pad and waits until it comes to rest. Yields where it landed and how high it flew.")]
    public static IEnumerator LaunchBall()
    {
        var j = _instance;
        var b = j.ball;
        var top = j.ledge.position.y + j.ledge.localScale.y * 0.5f;
        var front = j.ledge.position.z - j.ledge.localScale.z * 0.5f;
        b.isKinematic = true;
        b.position = j.pad.position + Vector3.up * 0.42f;
        b.transform.position = b.position;
        b.rotation = Quaternion.identity;
        j.SetGoal(Idle);
        j.trail.Clear();
        yield return null;
        yield return new WaitForSeconds(0.15f);
        var v = LaunchVelocity();
        SaturaLog.Log("Pad", $"launch velocity up {v.y:0.0} m/s, forward {v.z:0.0} m/s");
        b.isKinematic = false;
        b.linearDamping = 0f;
        b.linearVelocity = v;
        b.angularVelocity = Vector3.zero;
        var apex = b.position.y;
        var wallHit = -1f;
        var t = 0f;
        while (t < 4f)
        {
            t += Time.deltaTime;
            apex = Mathf.Max(apex, b.position.y);
            if (wallHit < 0f && b.position.z > front - 0.45f && b.position.y < top) wallHit = b.position.y;
            if (b.position.y > top && b.position.z > front) b.linearDamping = 2.5f;
            if (wallHit >= 0f && b.position.y < 0.5f) b.linearDamping = 1.5f;
            if (t > 0.6f && b.linearVelocity.magnitude < 0.3f) break;
            if (t > 0.5f && b.position.y < top + 0.45f && b.position.z > front && b.linearVelocity.y <= 0f) break;
            if (wallHit >= 0f && b.position.y < 0.6f) break;
            yield return null;
        }
        var onLedge = b.position.y > top - 0.1f && b.position.z > front;
        SaturaLog.Log("Pad", $"apex {apex - 0.35f:0.0} m, the ledge top is {top:0.0} m");
        if (onLedge) SaturaLog.Log("Pad", $"landed on the ledge, {b.position.z - front:0.0} m past the edge");
        else if (wallHit >= 0f) SaturaLog.Log("Pad", $"hit the ledge wall at {wallHit - 0.35f:0.0} m and fell back to the floor");
        else SaturaLog.Log("Pad", "fell short of the ledge");
        j.SetGoal(onLedge ? Hit : Miss);
        yield return new { landed = onLedge ? "ledge" : "floor", apex = Mathf.Round((apex - 0.35f) * 10f) / 10f, ledgeTop = Mathf.Round(top * 10f) / 10f };
    }
}
